       // --- 逻辑控制 ---                // 页面切换函数        function switchScreen(screenId) {            document.querySelectorAll('.screen').forEach(s => {                s.classList.remove('active');                s.classList.add('hidden');            });            const target = document.getElementById(screenId);            target.classList.remove('hidden');            target.classList.add('active');        }        function openChat() {            switchScreen('chat-screen');            // 如果是第一次进入，触发初始对话            const msgContainer = document.getElementById('messages');            if (msgContainer.children.length === 0) {                setTimeout(() => addMessage("我的左耳听到了你的告白", 'received'), 500);                setTimeout(() => addMessage("想要与我共度良宵吗？", 'received', true), 1500);            }        }        function goHome() {            switchScreen('home-screen');        }        function showProfile() {            switchScreen('profile-screen');        }        // 聊天功能        function addMessage(text, type, showChoices = false) {            const container = document.getElementById('messages');            const msgDiv = document.createElement('div');            msgDiv.className = `message ${type === 'sent' ? 'msg-sent' : 'msg-received'}`;            msgDiv.innerText = text;            container.appendChild(msgDiv);            container.scrollTop = container.scrollHeight;            if (showChoices) {                const choicesDiv = document.createElement('div');                choicesDiv.className = 'choices-area';                choicesDiv.innerHTML = `                    <button class="choice-btn" onclick="handleChoice('yes')">是</button>                    <button class="choice-btn" onclick="handleChoice('no')">否</button>                `;                container.appendChild(choicesDiv);                container.scrollTop = container.scrollHeight;            }        }        function sendMessage() {            const input = document.getElementById('msg-input');            const text = input.value.trim();            if (!text) return;                        addMessage(text, 'sent');            input.value = '';            // 简单回复逻辑            setTimeout(() => {                addMessage("我在听。", 'received');            }, 1000);        }        function handleChoice(choice) {            // 移除选项按钮            const choices = document.querySelector('.choices-area');            if (choices) choices.remove();            if (choice === 'yes') {                addMessage("是", 'sent');                setTimeout(() => {                    addMessage("我的荣幸", 'received');                    setTimeout(() => {                        showProfile(); // 跳转到档案页                    }, 1500);                }, 800);            } else {                addMessage("否", 'sent');                setTimeout(() => {                    addMessage("或许了解完我你会转变想法？", 'received');                    setTimeout(()=>{                        showProfile();//跳转到档案页                    },1500);                }, 800);            }        }        // 绑定回车发送        document.getElementById('msg-input').addEventListener('keypress', function (e) {            if (e.key === 'Enter') sendMessage();        });        // --- 背景粒子动画 ---        const canvas = document.getElementById('canvas-bg');        const ctx = canvas.getContext('2d');        let particles = [];        function resizeCanvas() {            canvas.width = window.innerWidth;            canvas.height = window.innerHeight;        }        window.addEventListener('resize', resizeCanvas);        resizeCanvas();        class Particle {            constructor() {                this.x = Math.random() * canvas.width;                this.y = Math.random() * canvas.height;                this.size = Math.random() * 2;                this.speedX = Math.random() * 0.5 - 0.25;                this.speedY = Math.random() * 0.5 - 0.25;                this.opacity = Math.random() * 0.5;            }            update() {                this.x += this.speedX;                this.y += this.speedY;                if (this.x > canvas.width) this.x = 0;                if (this.x < 0) this.x = canvas.width;                if (this.y > canvas.height) this.y = 0;                if (this.y < 0) this.y = canvas.height;            }            draw() {                ctx.fillStyle = `rgba(229, 57, 53, ${this.opacity})`;                ctx.beginPath();                ctx.arc(this.x, this.y, this.size, 0, Math.PI * 2);                ctx.fill();            }        }        function initParticles() {            for (let i = 0; i < 50; i++) {                particles.push(new Particle());            }        }        function animateParticles() {            ctx.clearRect(0, 0, canvas.width, canvas.height);            particles.forEach(p => {                p.update();                p.draw();            });            requestAnimationFrame(animateParticles);        }        initParticles();        animateParticles();    </script></body></html>
